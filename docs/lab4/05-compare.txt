# Об'єкти лаби 3 (k8s/) та чарта (helm template app chart/devops-lab)
## k8s/
kind: Namespace	  name: devops-lab
kind: ConfigMap	  name: app-config
kind: Secret	  name: app-secret
kind: PersistentVolumeClaim	  name: postgres-data
kind: Deployment	  name: postgres
kind: Service	  name: postgres
kind: Deployment	  name: backend
kind: Service	  name: backend
kind: Deployment	  name: frontend
kind: Service	  name: frontend
kind: Ingress	  name: devops-lab

kind: Secret	  name: app-devops-lab-secret
kind: ConfigMap	  name: app-devops-lab-config

kind: Service	  name: app-devops-lab-postgres
